Estrategia Pine Script v5 para TradingView (BTCUSD Engine)

Copia y pega este script en el editor Pinescript de TradingView: 
//@version=5
strategy("Quant Macro & Risk Engine - BTCUSD (VSA + S/R + EMA Cloud)", overlay=true, initial_capital=1000, default_qty_type=strategy.percent_of_equity, default_qty_value=100)

// --- INPUTS & PARAMETROS ---
leftBars = input.int(15, "Barras Izquierda Pivote")
rightBars = input.int(15, "Barras Derecha Pivote")
vsaVolMult = input.float(1.8, "Multiplicador Volumen VSA Ballena")
riskRatio = input.float(2.0, "Ratio Target Take Profit (1:N)")

// --- EMA TREND CLOUD ---
ema20 = ta.ema(close, 20)
ema50 = ta.ema(close, 50)
ema200 = ta.ema(close, 200)

plot(ema20, "EMA 20", color=color.rgb(14, 203, 129, 20), linewidth=2)
plot(ema50, "EMA 50", color=color.rgb(240, 185, 11, 20), linewidth=2)
plot(ema200, "EMA 200", color=color.rgb(246, 70, 93, 20), linewidth=2)

// --- VWAP INSTITUCIONAL ---
vwapVal = ta.vwap(hlc3)
plot(vwapVal, "VWAP Institucional", color=color.yellow, linewidth=2, style=plot.style_line)

// --- S/R PIVOTS DYNAMIC ---
pivotHigh = ta.pivothigh(high, leftBars, rightBars)
pivotLow = ta.pivotlow(low, leftBars, rightBars)

var float resLevel = na
var float supLevel = na

if not na(pivotHigh)
    resLevel := pivotHigh
if not na(pivotLow)
    supLevel := pivotLow

plot(resLevel, "Resistencia Pivot", color=color.red, linewidth=2, style=plot.style_line)
plot(supLevel, "Soporte Pivot", color=color.green, linewidth=2, style=plot.style_line)

// --- VSA VOLUME ANALYSIS ---
avgVol = ta.sma(volume, 20)
isHighVol = volume > (avgVol * vsaVolMult)
isBullishAbs = isHighVol and (close > open) and (close > vwapVal)
isBearishClimax = isHighVol and (close < open) and (close < vwapVal)

// --- SIGNAL GENERATION ---
longCondition = isBullishAbs and (close > supLevel) and (ema20 > ema50)
shortCondition = isBearishClimax and (close < resLevel) and (ema20 < ema50)

plotshape(longCondition, title="Buy Signal", style=shape.triangleup, location=location.belowbar, color=color.green, size=size.small, text="VSA BUY")
plotshape(shortCondition, title="Sell Signal", style=shape.triangledown, location=location.abovebar, color=color.red, size=size.small, text="VSA SELL")

// --- RISK & EXIT ENGINE (TP/SL) ---
if (longCondition)
    sl = supLevel * 0.998
    tp = close + ((close - sl) * riskRatio)
    strategy.entry("Long", strategy.long)
    strategy.exit("TP/SL Long", "Long", stop=sl, limit=tp)

if (shortCondition)
    sl = resLevel * 1.002
    tp = close - ((sl - close) * riskRatio)
    strategy.entry("Short", strategy.short)
    strategy.exit("TP/SL Short", "Short", stop=sl, limit=tp)
